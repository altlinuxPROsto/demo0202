! ===== HQ-RTR (EcoRouter) — Задание 2, стенд преподавателя =====
! Адреса стенда: isp 172.16.1.2, HQ-SRV 192.168.0.2 (интерфейс hq.100)
! Если HQ-SRV у вас с другим адресом — замените 192.168.0.2 ниже.
enable
configure terminal
! 8080 снаружи -> веб-приложение HQ-SRV:80
ip nat source static tcp 192.168.0.2 80 172.16.1.2 8080
! 2027 снаружи -> SSH HQ-SRV:2027
ip nat source static tcp 192.168.0.2 2027 172.16.1.2 2027
end
write memory
!
! Проверка: show ip nat static
! С ISP: curl -I http://172.16.1.2:8080   и   ssh -p 2027 sshuser@172.16.1.2
