! ===== HQ-RTR: переезд VLAN 100 с 192.168.100.0/27 на 192.168.1.0/27 =====
! Нужен, только если hq-rtr-1.txt уже вводили со старой сетью 192.168.100.x.
! Если какая-то строка "no ..." ругается, что такого нет, — это нормально, идём дальше.
enable
configure terminal
!
interface vl100
 no ip address 192.168.100.1/27
 ip address 192.168.1.1/27
exit
!
router ospf 1
 no network 192.168.100.0/27 area 0
 network 192.168.1.0/27 area 0
exit
!
no ip nat source dynamic inside-to-outside pool HQ overload interface isp
no ip nat pool HQ
ip nat pool HQ 192.168.1.1-192.168.1.30,192.168.200.1-192.168.200.14,192.168.99.1-192.168.99.6
ip nat source dynamic inside-to-outside pool HQ overload interface isp
!
dhcp-server 1
 pool CLI 1
  no dns 192.168.100.2
  dns 192.168.1.2
 exit
exit
!
! если уже делали Задание 2 со старыми адресами:
no ip nat source static tcp 192.168.100.2 80 172.16.1.2 8080
no ip nat source static tcp 192.168.100.2 2027 172.16.1.2 2027
ip nat source static tcp 192.168.1.2 80 172.16.1.2 8080
ip nat source static tcp 192.168.1.2 2027 172.16.1.2 2027
!
end
write memory
!
! Проверка: show ip interface brief | show ip ospf neighbor | show running-config
