! ===== BR-RTR (EcoRouter) — Задание 2, стенд преподавателя =====
! Адреса стенда: isp 172.16.2.2, BR-SRV 192.168.4.2 (за BR-FW)
! Если BR-SRV у вас с другим адресом — замените 192.168.4.2 ниже.
enable
configure terminal
! 8080 снаружи -> testapp (docker) BR-SRV:8080
ip nat source static tcp 192.168.4.2 8080 172.16.2.2 8080
! 2027 снаружи -> SSH BR-SRV:2027
ip nat source static tcp 192.168.4.2 2027 172.16.2.2 2027
! NTP-клиент -> ISP
ntp server 172.16.2.1
end
write memory
!
! Проверка: show ip nat static | show ntp status
! С ISP: curl -I http://172.16.2.2:8080   и   ssh -p 2027 sshuser@172.16.2.2
